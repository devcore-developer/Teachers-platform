import type { NextConfig } from "next";

/**
 * Phase 12 — production hardening configuration.
 *
 * Security headers are applied to every route. The Content-Security-Policy
 * is intentionally pragmatic: the App Router ships inline flight/bootstrap
 * scripts that require 'unsafe-inline', and dev mode requires 'unsafe-eval'
 * for React refresh. Everything else is locked to 'self' (no third-party
 * script/style/frame origins are used anywhere in the platform).
 *
 * frame-ancestors 'none' + X-Frame-Options DENY also protect the
 * certificate print sheet from clickjacking while keeping window.print().
 */
const isDev = process.env.NODE_ENV !== "production";

const csp = [
  "default-src 'self'",
  `script-src 'self' 'unsafe-inline'${isDev ? " 'unsafe-eval'" : ""}`,
  "style-src 'self' 'unsafe-inline'",
  "img-src 'self' data: blob:",
  "font-src 'self' data:",
  "connect-src 'self'",
  "frame-src 'none'",
  "object-src 'none'",
  "base-uri 'self'",
  "form-action 'self'",
  "frame-ancestors 'none'",
].join("; ");

const securityHeaders = [
  { key: "Content-Security-Policy", value: csp },
  { key: "X-Frame-Options", value: "DENY" },
  { key: "X-Content-Type-Options", value: "nosniff" },
  { key: "Referrer-Policy", value: "strict-origin-when-cross-origin" },
  { key: "X-DNS-Prefetch-Control", value: "off" },
  {
    key: "Permissions-Policy",
    value: "camera=(), microphone=(), geolocation=(), payment=(), usb=()",
  },
  // Only meaningful over HTTPS — ignored by browsers on plain HTTP dev.
  {
    key: "Strict-Transport-Security",
    value: "max-age=15552000; includeSubDomains",
  },
];

const nextConfig: NextConfig = {
  // Do not advertise the framework version.
  poweredByHeader: false,
  // Explicit bound on server-action bodies (the default is already 1MB;
  // making it explicit documents the zod CVE-2026-style DoS mitigation:
  // every z.array() in the validation layer is also .max()-bounded).
  experimental: {
    serverActions: {
      bodySizeLimit: "1mb",
    },
  },
  async headers() {
    return [
      { source: "/:path*", headers: securityHeaders },
    ];
  },
};

/**
 * Role-aware dashboard (server component). The session user comes from
 * the DB — never from the client:
 *   SUPER_ADMIN                -> platform dashboard + schools + users + academy metrics
 *   DEVELOPMENT_OFFICE_MANAGER -> same shell + THEIR school's metrics
 *                                 (schoolId from the session only)
 *   deputy / supervisor / teacher -> scoped role dashboard
 *
 * Phase 1 mock sections keep their honest "بيانات تجريبية" labels.
 */
export default async function HomePage() {
  const user = await requireUser();

  if (user.role === "SUPER_ADMIN") {
    return (
      <DashboardView
        schoolsSlot={
          <div className="space-y-6">
            <SchoolsMetricsSection />
            <UsersMetricsSection />
            <AcademyMetricsSection />
            <AssessmentMetricsSection />
          </div>
        }
      />
    );
  }

  if (user.role === "DEVELOPMENT_OFFICE_MANAGER" && user.schoolId) {
    return (
      <DashboardView
        schoolsSlot={
          <div className="space-y-6">
            <SchoolsMetricsSection schoolId={user.schoolId} />
            <UsersMetricsSection schoolId={user.schoolId} />
            <AcademyMetricsSection schoolId={user.schoolId} />
            <AssessmentMetricsSection schoolId={user.schoolId} />
          </div>
        }
      />
    );
  }

  return <RoleDashboard user={user} />;
}

export interface AuthFormState {
  ok: boolean;
  message?: string;
  fieldErrors?: Record<string, string>;
  /** Extra flag for the pending-verification screen (email not sent). */
  emailSent?: boolean;
  email?: string;
}

/** Derive the absolute app base URL from request headers (no env needed). */
async function appBaseUrl(): Promise<string> {
  const h = await headers();
  const host = h.get("x-forwarded-host") ?? h.get("host") ?? "localhost:3000";
  const proto = h.get("x-forwarded-proto") ?? "http";
  return `${proto}://${host}`;
}

function clientIp(h: Awaited<ReturnType<typeof headers>>): string {
  return h.get("x-forwarded-for")?.split(",")[0]?.trim() ?? "local";
}

/* ------------------------------- Login ---------------------------- */

export async function loginAction(
  _prevState: AuthFormState | null,
  formData: FormData
): Promise<AuthFormState> {
  const parsed = loginSchema.safeParse({
    email: String(formData.get("email") ?? ""),
    password: String(formData.get("password") ?? ""),
  });
  if (!parsed.success) {
    return {
      ok: false,
      message: "تحقق من الحقول المدخلة ثم أعد المحاولة",
      fieldErrors: zodAuthFieldErrors(parsed.error),
    };
  }

  const h = await headers();
  const ip = clientIp(h);
  if (isRateLimited(parsed.data.email, ip)) {
    const wait = retryAfterSeconds(parsed.data.email, ip);
    return {
      ok: false,
      message: `محاولات كثيرة خاطئة — حاول مجددًا بعد ${Math.ceil(wait / 60)} دقيقة تقريبًا`,
    };
  }

  const result = await authenticateCredentials(parsed.data.email, parsed.data.password);
  if (!result.ok) {
    if (result.code === "INVALID_CREDENTIALS") {
      recordLoginFailure(parsed.data.email, ip);
    }
    return { ok: false, message: result.message };
  }

  recordLoginSuccess(parsed.data.email, ip);
  await createSessionCookie(result.userId, result.tokenVersion);

  const next = sanitizeNextPath(String(formData.get("next") ?? ""));
  redirect(next);
}

/* ------------------------------- Logout --------------------------- */

export async function logoutAction(): Promise<void> {
  await destroySessionCookie();
  redirect("/login");
}

export async function registerAction(

  const result = await registerTeacher(
    {
      slug: String(formData.get("slug") ?? ""),
      name: String(formData.get("name") ?? ""),
      email: String(formData.get("email") ?? ""),
      password: String(formData.get("password") ?? ""),
      confirmPassword: String(formData.get("confirmPassword") ?? ""),
      branchId: String(formData.get("branchId") ?? ""),

import { cache } from "react";
import { cookies } from "next/headers";
import { redirect } from "next/navigation";

import { db } from "@/lib/db";
import {
  createSessionToken,
  verifySessionToken,
  SESSION_COOKIE,
  SESSION_TTL_SECONDS,
} from "@/lib/auth/token";

/* ------------------------------------------------------------------ */
/* Server-side session handling.                                        */
/*                                                                      */
/* The cookie is httpOnly + SameSite=Lax (+ Secure in production) and   */
/* holds a stateless JWT. Every authorization decision re-reads the     */
/* user from the database — role/activation/verification changes apply  */
/* on the very next request. localStorage is never a security boundary. */
/* ------------------------------------------------------------------ */

export { SESSION_COOKIE };

/** Session-safe user projection — passwordHash NEVER leaves this module. */
export interface SafeUser {
  id: string;
  name: string;
  email: string;
  role: string;
  schoolId: string | null;
  branchId: string | null;
  isActive: boolean;
  emailVerifiedAt: Date | null;
  tokenVersion: number;
}

/** Issue a session token for the user and set the auth cookie. */
export async function createSessionCookie(
  userId: string,
  tokenVersion: number
): Promise<void> {
  const token = await createSessionToken({ sub: userId, tv: tokenVersion });
  const store = await cookies();
  store.set(SESSION_COOKIE, token, {
    httpOnly: true,
    sameSite: "lax",
    secure: process.env.NODE_ENV === "production",
    path: "/",
    maxAge: SESSION_TTL_SECONDS,
  });
}

/** Clear the auth cookie (logout). */
export async function destroySessionCookie(): Promise<void> {
  const store = await cookies();
  store.delete(SESSION_COOKIE);
}

/**
 * Resolve the current signed-in user from the session cookie, with all
 * security state re-checked against the database. Returns null for any
 * failure: no cookie, invalid/expired token, unknown user, deactivated
 * account, or stale tokenVersion (revoked sessions).
 *
 * cache() keeps one DB read per request even when several components ask.
 */
export const getCurrentUser = cache(async (): Promise<SafeUser | null> => {
  try {
    const store = await cookies();
    const claims = await verifySessionToken(store.get(SESSION_COOKIE)?.value);
    if (!claims) return null;

    const user = await db.user.findUnique({ where: { id: claims.sub } });
    if (!user || !user.isActive) return null;
    if (user.tokenVersion !== claims.tv) return null;

    return {
      id: user.id,
      name: user.name,
      email: user.email,
      role: user.role,
      schoolId: user.schoolId,
      branchId: user.branchId,
      isActive: user.isActive,
      emailVerifiedAt: user.emailVerifiedAt,
      tokenVersion: user.tokenVersion,
    };
  } catch {
    // A missing AUTH_SECRET or a database hiccup means "not signed in"
    // rather than a crashed page; the provisioning/login flows surface
    // configuration errors explicitly.
    return null;
  }
});

/** Require a signed-in user or redirect to the Arabic login page. */
export async function requireUser(nextPath?: string): Promise<SafeUser> {
  const user = await getCurrentUser();
  if (!user) {
    const query = nextPath
      ? `?next=${encodeURIComponent(sanitizeNextPath(nextPath))}`
      : "";
    redirect(`/login${query}`);
  }
  return user;
}

/** Whitelist internal redirects coming from the ?next= parameter. */
export function sanitizeNextPath(value: string | undefined | null): string {
  if (value && value.startsWith("/") && !value.startsWith("//")) return value;
  return "/";
}

import bcrypt from "bcryptjs";

/* ------------------------------------------------------------------ */
/* Password hashing — bcrypt via bcryptjs (pure JS: no native build     */
/* steps on Windows/local machines). Plaintext passwords are never      */
/* stored, logged or embedded in URLs.                                  */
/* ------------------------------------------------------------------ */

const BCRYPT_COST = 12;

/** A fixed bcrypt hash of an unreachable random string. Compared against
 * when the account does not exist so that "user not found" and "wrong
 * password" take the same wall-clock time (timing-equalized login). */
const DUMMY_HASH = "$2b$12$C6UzMDM.H6dfI/f/IKcEe.5bTzDcZZowT8mSD2Fz.CbA9O7eO8fZu";

export function hashPassword(plain: string): Promise<string> {
  return bcrypt.hash(plain, BCRYPT_COST);
}

export function verifyPassword(
  plain: string,
  hash: string
): Promise<boolean> {
  return bcrypt.compare(plain, hash);
}

/** Timing-equalizing no-op comparison for unknown accounts. */
export function verifyDummyPassword(plain: string): Promise<boolean> {
  return bcrypt.compare(plain, DUMMY_HASH);
}

import { SignJWT, jwtVerify } from "jose";

/* ------------------------------------------------------------------ */
/* Session token (stateless JWT in an httpOnly cookie).                 */
/*                                                                      */
/* The token carries ONLY the user id and the account's tokenVersion —  */
/* role, activation and verification state are ALWAYS re-read from the  */
/* database on every request (src/lib/auth/session.ts), so revoking or  */
/* deactivating an account takes effect immediately.                    */
/*                                                                      */
/* This module is edge-safe: it uses `jose` (WebCrypto) and touches no  */
/* Node-only or next/headers APIs, so the middleware can verify tokens  */
/* without pulling the database layer into the edge runtime.            */
/* ------------------------------------------------------------------ */

export const SESSION_TTL_SECONDS = 60 * 60 * 24 * 7; // 7 days

/** Cookie name — defined here (edge-safe) so middleware and session.ts share it. */
export const SESSION_COOKIE = "mq_session";

export interface SessionClaims {
  /** User id. */
  sub: string;
  /** tokenVersion at sign time — mismatch invalidates the session. */
  tv: number;
}

function sessionSecret(): Uint8Array {
  const secret = process.env.AUTH_SECRET;
  if (!secret || secret.length < 32) {
    throw new Error(
      "AUTH_SECRET is missing or too short — set a 32+ char random value in .env"
    );
  }
  return new TextEncoder().encode(secret);
}

export async function createSessionToken(
  claims: SessionClaims,
  expiresAt: Date = new Date(Date.now() + SESSION_TTL_SECONDS * 1000)
): Promise<string> {
  return new SignJWT({ tv: claims.tv })
    .setProtectedHeader({ alg: "HS256" })
    .setSubject(claims.sub)
    .setIssuedAt()
    .setExpirationTime(expiresAt)
    .sign(sessionSecret());
}

/**
 * Verify a session token and return its claims, or null for ANY failure
 * (missing token, bad signature, expiry, missing/short AUTH_SECRET).
 * Never throws — callers treat null as "not signed in".
 */
export async function verifySessionToken(
  token: string | undefined | null
): Promise<SessionClaims | null> {
  if (!token) return null;
  try {
    const { payload } = await jwtVerify(token, sessionSecret(), {
      algorithms: ["HS256"],
    });
    if (!payload.sub || typeof payload.tv !== "number") return null;
    return { sub: payload.sub, tv: payload.tv };
  } catch {
    return null;
  }
}

/**
 * Phase 11 — account-security token flows.
 *
 * Contract (matches the schema comments, prisma/schema.prisma):
 *  - Raw tokens live ONLY inside the emailed URL. The database stores a
 *    SHA-256 hash (hash-at-rest), so a database leak cannot be replayed
 *    against the public endpoints.
 *  - Tokens are single-use (usedAt) and expiring (verification 24h,
 *    reset 1h).
 *  - Consuming a reset token bumps tokenVersion → every existing session
 *    JWT becomes invalid immediately.
 *  - Security emails are attempted IMMEDIATELY and never persisted with
 *    their token payload (see lib/notifications/emails.ts).
 *  - Password-reset requests NEVER reveal whether the email exists.

  if (!origin) return null;
  if (!path.startsWith("/")) return null;

export async function createEmailVerificationToken(userId: string): Promise<CreatedTokenResult> {

  if (!user) return { ok: false };

  // Keep at most one live token per purpose — replacing the old one

  await db.emailVerificationToken.deleteMany({

  const expiresAt = new Date(Date.now() + VERIFICATION_TOKEN_HOURS * 60 * 60 * 1000);
  await db.emailVerificationToken.create({

export type ConsumeTokenFailure = "INVALID" | "EXPIRED" | "USED";

export async function consumeEmailVerificationToken(
  rawToken: string,
): Promise<
  | { ok: true; user: { id: string; name: string; email: string; schoolId: string | null; role: string }; alreadyVerified: boolean }
  | { ok: false; reason: ConsumeTokenFailure }
> {
  if (!rawToken || rawToken.length < 20 || rawToken.length > 200) {
    return { ok: false, reason: "INVALID" };
  }
  const record = await db.emailVerificationToken.findUnique({
    where: { tokenHash: hashToken(rawToken) },
    select: { id: true, userId: true, expiresAt: true, usedAt: true },
  });
  if (!record) return { ok: false, reason: "INVALID" };
  if (record.usedAt) return { ok: false, reason: "USED" };
  if (record.expiresAt.getTime() <= Date.now()) return { ok: false, reason: "EXPIRED" };

  const user = await db.user.findUnique({
    where: { id: record.userId },
    select: { id: true, name: true, email: true, schoolId: true, role: true, isActive: true, emailVerifiedAt: true },
  });
  if (!user || !user.isActive) return { ok: false, reason: "INVALID" };

  const alreadyVerified = user.emailVerifiedAt !== null;

  // Mark used + set verification timestamp atomically.
  await db.$transaction([
    db.emailVerificationToken.update({
      where: { id: record.id },
      data: { usedAt: new Date() },
    }),
    ...(alreadyVerified
      ? []
      : [db.user.update({
          where: { id: user.id },
          data: { emailVerifiedAt: new Date() },
        })]),
  ]);

  // Activation event → in-app notification (+ queued welcome email for
  // the ACCOUNT category, which is never preference-gated).
  if (!alreadyVerified) {
    await notify({
      userId: user.id,
      type: "ACCOUNT_ACTIVATED",
      titleAr: "تم تفعيل حسابك بنجاح",
      bodyAr: "تم تأكيد بريدك الإلكتروني وتفعيل حسابك داخل المنصة. يمكنك الآن متابعة لوحتك التدريبية.",
      resourceType: "User",
      resourceId: user.id,
      destinationUrl: "/dashboard",
      eventId: `account-activated:${user.id}`,
      email: true,
    });
  }

  return {
    ok: true,
    user: { id: user.id, name: user.name, email: user.email, schoolId: user.schoolId, role: user.role },
    alreadyVerified,
  };
}

  accepted: true;
  /** Global, non-sensitive deployment fact — safe to display. */
  emailConfigured: boolean;
  emailSent: boolean;

/**
 * Phase 12 — per-email flood control for "forgot password".
 * An attacker must not be able to mail-bomb arbitrary addresses through
 * this public flow. Limits are deliberately low (3 / 15 min per email);
 * a blocked request returns the SAME anti-enumeration shape so the
 * response never becomes an account-existence oracle.
 */
const RESET_REQUEST_LIMIT = 3;
const RESET_REQUEST_WINDOW_MS = 15 * 60 * 1000;

/**
 * Handles a "forgot password" request. The outcome is IDENTICAL for
 * existing and non-existing emails — no account enumeration.
 */
export async function requestPasswordReset(rawEmail: string): Promise<PasswordResetRequestOutcome> {
  const email = rawEmail.trim().toLowerCase();
  if (!email || !/^[^@\s]+@[^@\s]+\.[^@\s]+$/.test(email)) {
    return { accepted: true, emailConfigured: false, emailSent: false, note: "INVALID_INPUT" };
  }

  // Rate limit BEFORE any DB lookup so the limiter applies uniformly and
  // cannot be used to probe which emails exist (identical outcome either way).
  const limiter = checkRateLimit(
    "password-reset-request",
    email,
    RESET_REQUEST_LIMIT,
    RESET_REQUEST_WINDOW_MS
  );
  if (!limiter.allowed) {
    console.warn(
      `[security] password-reset request rate-limited for an address (retry in ${limiter.retryAfterSeconds}s)`
    );
    return {
      accepted: true,
      emailConfigured: false,
      emailSent: false,
      rateLimited: true,
      note: "RATE_LIMITED",
    };
  }

  const user = await db.user.findUnique({
    where: { email },
    select: { id: true, name: true, isActive: true, schoolId: true },
  });

  if (!user || !user.isActive) {
    return { accepted: true, emailConfigured: false, emailSent: false };
  }

  const created = await createPasswordResetToken(user.id);
  if (!created.ok || !created.token) {
    return { accepted: true, emailConfigured: false, emailSent: false };
  }
  if (!created.actionUrl) {
    // Deployment without APP_PUBLIC_ORIGIN — an absolute reset link cannot
    // be built, so no misleading email is sent. Recorded as NOT_CONFIGURED.
    return {
      accepted: true,
      emailConfigured: false,
      emailSent: false,
      note: "APP_PUBLIC_ORIGIN_MISSING",
    };
  }

  const outcome = await sendSecurityEmailNow({
    toEmail: email,
    userId: user.id,
    schoolId: user.schoolId,
    templateKey: "password-reset",
    params: {
      recipientName: user.name,
      resetUrl: created.actionUrl,
    },
  });

  if (outcome.ok) return { accepted: true, emailConfigured: true, emailSent: true };
  if ("configured" in outcome) {
    return { accepted: true, emailConfigured: false, emailSent: false, note: "EMAIL_NOT_CONFIGURED" };
  }
  return {
    accepted: true,
    emailConfigured: true,
    emailSent: false,
    note: `EMAIL_ERROR:${outcome.errorCategory}`,
  };
}

export async function consumePasswordResetToken(
  rawToken: string,
  newPassword: string,
): Promise<ResetPasswordResult> {
  if (!validateNewPassword(newPassword)) return { ok: false, reason: "WEAK_PASSWORD" };

  const record = await db.passwordResetToken.findUnique({
    where: { tokenHash: hashToken(rawToken) },
    select: { id: true, userId: true, expiresAt: true, usedAt: true },
  });
  if (!record) return { ok: false, reason: "INVALID" };
  if (record.usedAt) return { ok: false, reason: "USED" };
  if (record.expiresAt.getTime() <= Date.now()) return { ok: false, reason: "EXPIRED" };

  const user = await db.user.findUnique({
    where: { id: record.userId },
    select: { id: true, name: true, email: true, isActive: true, schoolId: true },
  });
  if (!user || !user.isActive) return { ok: false, reason: "INVALID" };

  const passwordHash = await hashPassword(newPassword);
  const now = new Date();

  // Atomic: consume token + rotate password + invalidate all sessions.
  await db.$transaction([
    db.passwordResetToken.update({ where: { id: record.id }, data: { usedAt: now } }),
    db.user.update({
      where: { id: user.id },
      data: { passwordHash, tokenVersion: { increment: 1 } },
    }),
  ]);

  // Security notice — in-app (always) + immediate email (never gated).
  await notify({
    userId: user.id,
    type: "PASSWORD_CHANGED",
    titleAr: "تم تغيير كلمة مرور حسابك",
    bodyAr: `تم تغيير كلمة مرور حسابك بتاريخ ${formatEmailDate(now)}، وتم إنهاء جميع الجلسات النشطة حفاظًا على الأمان. إن لم تقم أنت بهذا التغيير فتواصل مع إدارة المدرسة فورًا.`,
    resourceType: "User",
    resourceId: user.id,
    destinationUrl: "/notifications/preferences",
    eventId: `password-changed:${record.id}`,
    email: false, // email part is sent immediately below (token-free).
  }).catch(() => null);
  await sendSecurityEmailNow({
    toEmail: user.email,
    userId: user.id,
    schoolId: user.schoolId,
    templateKey: "password-changed",
    params: { recipientName: user.name, dateText: formatEmailDate(now) },
  }).catch(() => null);

  return { ok: true, userId: user.id };
}

export function safeSecretEquals(a: string, b: string): boolean {

/* ------------------------------------------------------------------ */
/* In-memory login rate limiting.                                       */
/*                                                                      */
/* Reasonable brute-force protection for a single-server deployment:    */
/* per-email sliding window. Documented limitations (not hidden):       */
/*   1. Counters live in process memory — they reset on server restart  */
/*      and are not shared across multiple instances.                   */
/*   2. A distributed deployment should swap this module for a          */
/*      Redis/DB-backed limiter without touching call-sites.            */
/* ------------------------------------------------------------------ */

interface AttemptRecord {
  failures: number[];
  lockedUntil: number;
}

const MAX_FAILURES = 5;
const WINDOW_MS = 15 * 60 * 1000; // 15 minutes
const LOCKOUT_MS = 15 * 60 * 1000; // 15 minutes

const records = new Map<string, AttemptRecord>();

function keyFor(email: string, ip: string): string {
  return `${email.toLowerCase()}|${ip}`;
}

function prune(record: AttemptRecord, now: number) {
  record.failures = record.failures.filter((t) => now - t < WINDOW_MS);
}

/** True when the caller must be blocked right now. */
export function isRateLimited(email: string, ip: string): boolean {
  const record = records.get(keyFor(email, ip));
  if (!record) return false;
  const now = Date.now();
  if (record.lockedUntil > now) return true;
  prune(record, now);
  return record.failures.length >= MAX_FAILURES;
}

/** Seconds until the lockout lifts (for the Arabic message). */
export function retryAfterSeconds(email: string, ip: string): number {
  const record = records.get(keyFor(email, ip));
  if (!record) return 0;
  return Math.max(0, Math.ceil((record.lockedUntil - Date.now()) / 1000));
}

/** Record a failed attempt; locks the key once the window is exhausted. */
export function recordLoginFailure(email: string, ip: string): void {
  const key = keyFor(email, ip);
  const now = Date.now();
  const record = records.get(key) ?? { failures: [], lockedUntil: 0 };
  prune(record, now);
  record.failures.push(now);
  if (record.failures.length >= MAX_FAILURES) {
    record.lockedUntil = now + LOCKOUT_MS;
    record.failures = [];
  }
  records.set(key, record);
}

/** Record a successful login — clears the failure counter. */
export function recordLoginSuccess(email: string, ip: string): void {
  records.delete(keyFor(email, ip));
}

/** Test helper: clear all counters. */
export function resetRateLimiter(): void {
  records.clear();
}

/**
 * Phase 12 — shared lightweight rate limiting (moved from
 * lib/surveys/rate-limit.ts so every sensitive flow can use it).
 *
 * Process-local sliding in-memory window keyed by `scope:subject`.
 * Best-effort by nature: limits reset on process restart and are
 * per-instance. For this platform's scale (single Next.js process,
 * SQLite dev phase) it stops burst abuse of public and sensitive
 * write paths (survey submissions, password-reset mail, certificate
 * verification) without adding infrastructure.
 */

interface Bucket {
  timestamps: number[];
}

const buckets = new Map<string, Bucket>();

/** Periodic sweep so abandoned buckets don't grow unbounded. */
const SWEEP_INTERVAL_MS = 300_000;
let lastSweep = 0;

function sweep(now: number, windowMs: number): void {
  if (now - lastSweep < SWEEP_INTERVAL_MS) return;
  lastSweep = now;
  for (const [key, bucket] of buckets) {
    bucket.timestamps = bucket.timestamps.filter((ts) => now - ts < windowMs);
    if (bucket.timestamps.length === 0) buckets.delete(key);
  }
}

export interface RateLimitResult {
  allowed: boolean;
  /** Seconds until the next attempt is allowed (when blocked). */
  retryAfterSeconds: number;
}

export function checkRateLimit(
  scope: string,
  subject: string,
  limit: number,
  windowMs: number
): RateLimitResult {
  const now = Date.now();
  sweep(now, windowMs);
  const key = `${scope}:${subject}`;
  const bucket = buckets.get(key) ?? { timestamps: [] };
  bucket.timestamps = bucket.timestamps.filter((ts) => now - ts < windowMs);

  if (bucket.timestamps.length >= limit) {
    const oldest = bucket.timestamps[0] ?? now;
    const retryAfterSeconds = Math.max(1, Math.ceil((windowMs - (now - oldest)) / 1000));
    buckets.set(key, bucket);
    return { allowed: false, retryAfterSeconds };
  }

  bucket.timestamps.push(now);
  buckets.set(key, bucket);
  return { allowed: true, retryAfterSeconds: 0 };
}

/** Test helper — clears all buckets (used by the integration suites). */
export function resetRateLimits(): void {
  buckets.clear();
}

/**
 * Best-effort client identity for rate limiting unauthenticated flows.
 * Reads the standard proxy headers; falls back to "unknown" so the
 * limiter still applies (one shared bucket) when no proxy headers
 * exist. Never used for authorization — rate limiting only.
 */
export function clientKeyFromHeaders(headers: Headers): string {
  const forwarded = headers.get("x-forwarded-for");
  if (forwarded) {
    const first = forwarded.split(",")[0]?.trim();
    if (first) return first;
  }
  return headers.get("x-real-ip")?.trim() || "unknown";
}

/**
 * Phase 11 — optional in-process scheduler (FALLBACK ONLY).
 *
 * Production deployments should drive background work through an EXTERNAL
 * scheduler calling the secured POST /api/cron/dispatch endpoint with the
 * CRON_SECRET header — an in-memory timer cannot be relied upon for
 * production scheduling (process restarts, multi-instance deployments).
 *
 * This instrumentation only runs when ENABLE_INTERNAL_SCHEDULER=true,
 * never registers twice, and unrefs its timer so it never keeps the
 * process alive. Every tick is the same idempotent code path the cron
 * endpoint uses.
 */

export async function register(): Promise<void> {
  if (process.env.NEXT_RUNTIME !== "nodejs") return;
  if ((process.env.ENABLE_INTERNAL_SCHEDULER ?? "").trim() !== "true") return;

  const globalRef = globalThis as typeof globalThis & {
    __mqPhase11SchedulerTimer?: NodeJS.Timeout;
  };
  if (globalRef.__mqPhase11SchedulerTimer) return;

  const { INTERNAL_SCHEDULER_INTERVAL_SECONDS } = await import("@/config/notifications");
  const { scanAssignmentDeadlineReminders, scanCourseCompletions } = await import(
    "@/lib/notifications/reminders"
  );
  const { processEmailOutbox } = await import("@/lib/notifications/dispatcher");

  const tick = async (): Promise<void> => {
    try {
      await scanAssignmentDeadlineReminders();
    } catch (error) {
      // Logged but never fatal — the next tick retries idempotently.
      console.error(
        "[scheduler] deadline scan failed:",
        error instanceof Error ? error.message : error
      );
    }
    try {
      await scanCourseCompletions();
    } catch (error) {
      console.error(
        "[scheduler] course-completion scan failed:",
        error instanceof Error ? error.message : error
      );
    }
    try {
      await processEmailOutbox();
    } catch (error) {
      console.error(
        "[scheduler] email dispatch failed:",
        error instanceof Error ? error.message : error
      );
    }
  };

  const timer = setInterval(() => {
    void tick();
  }, Math.max(30, INTERNAL_SCHEDULER_INTERVAL_SECONDS) * 1000);
  timer.unref?.();
  globalRef.__mqPhase11SchedulerTimer = timer;
}

// ─────────────────────────────────────────────────────────────────
// Phase 8 — Certificates & professional growth.
// Certificates are ALWAYS school-owned (schoolId NOT NULL): a
// certificate records that ONE teacher of THIS school satisfied the
// completion requirements of a course/program, per the SAME rules the
// server-side eligibility engine enforces at issuance time.
//
// Status model: ISSUED | REVOKED persisted. Expiration is DERIVED
// from expiresAt (never a stored status), so a certificate can never
// be simultaneously revoked and expired.
//
// Duplicate prevention: unique (teacherId, courseId) and
// (teacherId, programId) at the DB level — one completion
// certificate per teacher per qualifying content. Rows remain after
// revocation (audit-first), so a revoked certificate still blocks
// accidental re-issuance.
// ─────────────────────────────────────────────────────────────────

model Certificate {
  id       String @id @default(cuid())
  /// Owning school tenant (the ISSUING school — never null).
  schoolId String
  school   School @relation(fields: [schoolId], references: [id], onDelete: Restrict)

  /// The certified teacher. Always an active TEACHER of the owning
  /// school (service-enforced on every write — never client-trusted).
  teacherId String
  teacher   User   @relation("CertificateOwner", fields: [teacherId], references: [id], onDelete: Restrict)

  /// Human-readable serial (MQ-YYYY-XXXXXX) — unique, printed on the
  /// certificate and accepted by the verification search.
  certificateNumber String @unique
  /// Public verification code (16-char Crockford base32, stored
  /// WITHOUT dashes) — unique, collision-resistant, safe to print.
  verificationCode String @unique

  titleAr       String
  titleEn       String?
  descriptionAr String?
  descriptionEn String?

  /// The qualifying content. The service requires at least ONE of the
  /// two — eligibility is always computed against real learning data.
  programId String?
  program   TrainingProgram? @relation(fields: [programId], references: [id], onDelete: SetNull)
  courseId  String?
  course    Course?          @relation(fields: [courseId], references: [id], onDelete: SetNull)

  /// ISSUED | REVOKED — canonical key, validated with Zod on write.
  status String @default("ISSUED")

  issuedAt  DateTime
  /// Null = the certificate never expires.
  expiresAt DateTime?

  /// Issuance audit: who issued it + issuing-authority label snapshot.
  issuedById      String
  issuedBy        User    @relation("CertificateIssuer", fields: [issuedById], references: [id])
  issuerRoleLabel String?

  /// Revocation audit (null while ISSUED).
  revokedById      String?
  revokedBy        User?     @relation("CertificateRevoker", fields: [revokedById], references: [id])
  revokedAt        DateTime?
  revocationReason String?

  /// JSON snapshot of the eligibility rules + achieved results checked
  /// server-side at issuance (audit of WHAT qualified the teacher).
  criteriaSnapshotJson String?

  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt

  events CertificateEvent[]
  /** Phase 9 — impact surveys linked to this certificate. */
  surveys Survey[]

  /// One completion certificate per teacher per qualifying content.
  @@unique([teacherId, courseId])
  @@unique([teacherId, programId])
  @@index([schoolId, status])
  @@index([teacherId])
  @@map("certificates")
}

/**
 * Append-only audit trail for certificate lifecycle events (ISSUED /

 */
model CertificateEvent {
  id            String      @id @default(cuid())

  type String
  /// Actor that performed the action (null = system).
  actorId   String?

  /// Role key snapshot at event time.
  actorRole String?
  /// Free-text note (the revocation reason is duplicated here).
  note      String?
  createdAt DateTime @default(now())

  @@index([certificateId])
  @@map("certificate_events")
}

/* Tenancy: surveys and follow-ups are ALWAYS school-owned (schoolId     */

/**

 */
model Survey {
  id       String @id @default(cuid())
  /// Owning school tenant (never null — surveys are school work).
  schoolId String

  /// Authoring user (office manager or supervisor within their scope).
  ownerUserId String

  titleEn        String?
  descriptionAr  String?
  descriptionEn  String?
  /// Purpose shown to respondents before the questions.

  // (validated in the service on every write).
  programId     String?
  program       TrainingProgram? @relation(fields: [programId], references: [id], onDelete: SetNull)
  courseId      String?
  course        Course?          @relation(fields: [courseId], references: [id], onDelete: SetNull)
  lessonId      String?
  lesson        CourseLesson?    @relation(fields: [lessonId], references: [id], onDelete: SetNull)

  certificate   Certificate?     @relation(fields: [certificateId], references: [id], onDelete: SetNull)

  /// PRE_TRAINING | AFTER_TRAINING | APPLICATION_FOLLOWUP | null
  /// (null = a general survey with no impact-evaluation stage).
  stage String?
  /// Shared pairing code joining a pre-training survey with its
  /// after-training twin (e.g. "P9-PAIR-007"). Required for stage

// ─────────────────────────────────────────────────────────────────
// Phase 11 — notifications, preferences & email delivery audit.
//
// Recipient ownership: a notification row is PRIVATE to its user —
// every read path filters on userId server-side (never client-trusted).
//
// Idempotency: event-driven notifications carry a stable eventId and
// the DB enforces unique (userId, eventId), so retrying the same event
// can never duplicate a notification. NULL eventIds (rare, one-shot
// writes) are unconstrained by SQLite's NULL-aware unique semantics.
//
// Destination safety: destinationUrl always starts with a single "/"
// and is validated against the internal path allow-list in
// src/lib/notifications/urls.ts — external URLs are never stored.
//
// Email: delivery attempts are recorded in EmailDelivery (queue +
// audit in one row) with provider message ids, attempt counts and
// SANITIZED error categories only — no tokens, no message bodies,
// no provider secrets are ever persisted here.
// ─────────────────────────────────────────────────────────────────

/**
 * One in-app notification for ONE recipient. Arabic is the primary
 * language; optional English mirrors exist for bilingual rendering.
 * readAt is set by the owner's own actions only (ownership enforced
 * in the service layer and API routes).
 */
model Notification {
  id        String   @id @default(cuid())
  userId    String
  user      User     @relation("NotificationOwner", fields: [userId], references: [id], onDelete: Cascade)

  /// Canonical event key (e.g. ASSIGNMENT_PUBLISHED) — see
  /// src/config/notifications.ts for the full list.
  type      String
  /// ASSIGNMENTS | ASSESSMENTS | COURSES | SURVEYS | FOLLOWUPS |
  /// CERTIFICATES | ACCOUNT | PLATFORM — drives preference filtering.
  category  String

  titleAr   String
  bodyAr    String
  titleEn   String?
  bodyEn    String?

  /// Related resource discriminator (e.g. "Assignment") + id — used for
  /// dedupe context and future deep-links; never trusted for access.
  resourceType String?
  resourceId   String?
  /// Validated INTERNAL destination (starts with "/") or null.
  destinationUrl String?

  /// Mirror of the linked email delivery status (null = no email was
  /// part of this notification): QUEUED | SENT | FAILED | DISABLED.
  emailStatus String?

  /// Stable event identifier for idempotency (unique per recipient).
  eventId   String?

  readAt    DateTime?
  createdAt DateTime @default(now())

  @@unique([userId, eventId])
  @@index([userId, createdAt])
  @@index([userId, readAt])
  @@index([category, createdAt])
  @@map("notifications")
}

/**
 * Per-user delivery preferences. Persisted and enforced SERVER-SIDE:
 * the notification service consults this row before creating anything.
 * Account-security messages (email verification, password reset,
 * password changed) are NOT governed by these flags — they always
 * proceed because secure operation depends on them.
 */
model NotificationPreference {

  assignments Boolean @default(true)

  surveys     Boolean @default(true)
  followups   Boolean @default(true)
  certificates Boolean @default(true)
  platform    Boolean @default(true)

  updatedAt DateTime @updatedAt

  @@map("notification_preferences")
}

/**
 * Queue + audit row for ONE outbound email. Lifecycle:
 *   QUEUED -> PROCESSING -> SENT | FAILED
 * with bounded retries (attemptCount/maxAttempts + nextAttemptAt
 * backoff) and optimistic claiming (conditional updateMany on status)
 * so concurrent workers can never double-send the same row.
 *
 * Privacy: subject/template/category are operational metadata; raw
 * message bodies, verification tokens and provider secrets are never
 * stored. Recipient emails are masked on every read path exposed to
 * the UI (local-part asterisked).
 */
model EmailDelivery {
  id         String   @id @default(cuid())
  toEmail    String
  toUserId   String?
  user       User?    @relation("EmailDeliveryOwner", fields: [toUserId], references: [id], onDelete: SetNull)

  subject    String
  /// Template key (e.g. "assignment-published") — see templates.ts.
  template   String
  /// ACCOUNT (security/account mail) or a Notification category.
  category   String   @default("ACCOUNT")
  /// Owning school tenant when the mail is school-scoped (null =
  /// platform/account mail). Used for school-scoped delivery health.
  schoolId   String?
  notificationId String?

  /// QUEUED | PROCESSING | SENT | FAILED
  status     String   @default("QUEUED")
  provider   String?
  providerMessageId String?

  attemptCount  Int      @default(0)
  maxAttempts   Int      @default(4)
  nextAttemptAt DateTime @default(now())
  lockedAt      DateTime?
  lockedBy      String?

  /// Sanitized failure classification (NETWORK | AUTH | RATE_LIMIT |
  /// VALIDATION | PROVIDER | NOT_CONFIGURED) + short provider message.
  lastErrorCategory String?
  lastErrorMessage  String?
  lastAttemptAt     DateTime?
  sentAt            DateTime?

  /// Stable idempotency key — unique at the DB level so the same
  /// logical email can never be queued twice.
  dedupeKey  String?  @unique

  createdAt  DateTime @default(now())
  updatedAt  DateTime @updatedAt

  @@index([status, nextAttemptAt])
  @@index([schoolId, createdAt])
  @@index([toUserId, createdAt])
  @@map("email_deliveries")
}

/* ------------------------------------------------------------------ */
/* Manarat Qurtubah — development bootstrap seed.                       */
/*                                                                      */
/* The database ships EMPTY: /admin/accounts (which creates staff       */
/* accounts) is itself protected, so the first SUPER_ADMIN must come    */
/* from here. This script seeds one school tenant plus one account      */
/* per platform role so every phase can be explored immediately.        */
/*                                                                      */
/* Plain CommonJS on purpose: run it with `node prisma/seed.js` —       */
/* it needs NOTHING beyond the existing dependencies (@prisma/client    */
/* + bcryptjs, which is pure JS and builds natively nowhere).           */
/*                                                                      */
/* Idempotent: every record is an upsert (or checked before insert),   */
/* so re-running resets the demo accounts to the exact credentials      */
/* documented below without duplicating any rows.                       */
/*                                                                      */
/* These are DEVELOPMENT credentials — rotate them before any real      */
/* deployment.                                                          */

const { PrismaClient } = require("@prisma/client");
const bcrypt = require("bcryptjs");

const db = new PrismaClient();

/* Same cost as src/lib/auth/password.ts (BCRYPT_COST = 12). */
const BCRYPT_COST = 12;

const SCHOOL_SLUG = "al-nahda";
const BRANCH_NAME = "قسم البنين";

const ACCOUNTS = [

async function main() {
  console.log("Hashing passwords (bcrypt, cost 12)...");
  const seeded = [];
  for (const account of ACCOUNTS) {
    seeded.push({
      ...account,
      passwordHash: await bcrypt.hash(account.password, BCRYPT_COST),
    });
  }

  /* 1) School tenant ------------------------------------------------ */
  const school = await db.school.upsert({
    where: { slug: SCHOOL_SLUG },
    update: {},
    create: {
      nameAr: "مدرسة النهضة النموذجية",
      nameEn: "Al-Nahda Model School",
      slug: SCHOOL_SLUG,
    },
  });
  console.log(`School ready: ${school.nameAr} (${school.slug})`);

  /* 2) Home branch. nameNorm mirrors the service-layer normalization
     (trim + lowercase) required by the @@unique([schoolId, nameNorm]). */
  const nameNorm = BRANCH_NAME.trim().toLowerCase();
  const branch = await db.branch.upsert({
    where: { schoolId_nameNorm: { schoolId: school.id, nameNorm } },
    update: {},
    create: {
      schoolId: school.id,
      nameAr: BRANCH_NAME,
      nameEn: "Boys Section",
      nameNorm,
      stage: "primary",
    },
  });
  console.log(`Branch ready: ${branch.nameAr}`);

  /* 3) Users — one per role. Provisioned accounts arrive pre-verified
     (same policy as createStaffAccount); only self-registered teachers
     go through the email-verification flow. */
  const ids = {};
  for (const account of seeded) {
    const scoped = {
      schoolId: account.school ? school.id : null,
      branchId: account.branch ? branch.id : null,
    };
    const user = await db.user.upsert({
      where: { email: account.email },
      update: {
        name: account.name,
        passwordHash: account.passwordHash,
        role: account.role,
        ...scoped,
        isActive: true,
        emailVerifiedAt: new Date(),
      },
      create: {
        name: account.name,
        email: account.email,
        passwordHash: account.passwordHash,
        role: account.role,
        ...scoped,
        isActive: true,
        emailVerifiedAt: new Date(),
      },
    });
    ids[account.role] = user.id;
    console.log(`Account ready: ${account.email} -> ${account.role}`);
  }

  /* 4) Supervisor branch scope — what the supervisor may see. -------- */
  await db.supervisorAssignment.upsert({
    where: {
      supervisorId_branchId: {
        supervisorId: ids.EDUCATIONAL_SUPERVISOR,
        branchId: branch.id,
      },
    },
    update: {},
    create: {
      supervisorId: ids.EDUCATIONAL_SUPERVISOR,
      schoolId: school.id,
      branchId: branch.id,
    },
  });

  /* 5) Deputy read-only grant over the WHOLE school (branchId null).
     DeputyGrant has no unique constraint — check before inserting. --- */
  const existingGrant = await db.deputyGrant.findFirst({
    where: { userId: ids.EDUCATIONAL_DEPUTY, schoolId: school.id, branchId: null },
  });
  if (!existingGrant) {
    await db.deputyGrant.create({
      data: {
        userId: ids.EDUCATIONAL_DEPUTY,
        schoolId: school.id,
        branchId: null,
      },
    });
  }

  /* 6) Teacher -> supervisor assignment (@@unique on teacherId:
     one current supervisor at a time). ------------------------------- */
  await db.teacherSupervisorAssignment.upsert({
    where: { teacherId: ids.TEACHER },
    update: { supervisorId: ids.EDUCATIONAL_SUPERVISOR },
    create: {
      teacherId: ids.TEACHER,
      supervisorId: ids.EDUCATIONAL_SUPERVISOR,
      schoolId: school.id,
    },
  });

  console.log("\nSeed complete — sign in at /login with:");
  console.table(
    ACCOUNTS.map((account) => ({
      email: account.email,
      password: account.password,
      role: account.role,
    }))
  );
}

main()
  .catch((error) => {
    console.error("Seed failed:", error);
    process.exitCode = 1;
  })
  .finally(async () => {
    await db.$disconnect();
  });
